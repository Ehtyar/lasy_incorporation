\begin{frame}{Flying focus lasers -- solving the Problem of dephasing}{2. Axiparabola laser}
    \begin{columns}
        \begin{column}{0.5\textwidth}
        \begin{itemize}
            \item Axiparabola \cite{Smartsev:2019}:\begin{itemize}
            \item Near-parabolic mirror
            \item Focuses light onto a line -- the focus region
            \item[$\rightarrow$] Light at radius $r$ is focused at $f(r)=f_0+\delta\left(\frac{r}{R}\right)^2$
            \end{itemize}
            \item \textbf{R}adial \textbf{G}roup \textbf{D}elay echelon (RGD) \cite{Ambat:2023}\cite{Palastro:2020}:\begin{itemize}
            \item Stepped concentric mirror rings
            \item Shape follows some function $\tau_D(r)$
            \item[$\rightarrow$] controls the timing of the axiparabola focus
            \end{itemize}
        \end{itemize}
        {\scriptsize Images: Top: Axiparabola functionality. Image taken from Smartsev et al \cite{Smartsev:2019}. Bottom: RGD functionality. Image taken from Palastro et al \cite{Palastro:2020}.}

\begin{frame}{Implementing the flying focus}{The complete setup}
    The simulation had the following steps:
    \begin{enumerate}
        \item Initialising the laser pulse with its pulse profile
        \item Applying the RGD to the laser
        \item Applying the axiparabola to the laser
        \item Propagating the laser to the focus region\begin{itemize}
            \item At this point the laser field can be saved to file
            \item The next steps can happen in Lasy or PIConGPU
        \end{itemize}
        \item \label{measure_point} Measuring the relative focus position $t_{focus}-f_0/c$ or $z_{focus}-t\cdot c$ and the beam waist $w$
        \item \label{prop_point} Propagating the laser some distance through the focus region
        \item Repeating steps \ref{measure_point} and \ref{prop_point} until the end of the focus region is reached
    \end{enumerate}
\end{frame}

\section{Verifying the flying focus}
\frame{\sectionpage}
\addtocounter{framenumber}{-1}
\begin{frame}{Verifying the flying focus}{The measurement}
    \begin{columns}
        \begin{column}{0.55\textwidth}
        \begin{itemize}
            
            \item Relative focus position $t_{focus}-f_0/c$ (Lasy) or $z_{focus}-t\cdot c$ (PIConGPU):
            \begin{itemize}
                \item A central line-out is taken from the field energy, already in relative coordinates
                \item Then $t_{focus}-f_0/c$ or $z_{focus}-t\cdot c$ is calculated as the expected value of the resulting distribution
            \end{itemize}
            \item Beam waist $w$:
            \begin{itemize}
                \item The field energy is longitudinally and azimuthally summed
                \item Then $w$ is calculated from the statistical standard deviation of the resulting distribution
            \end{itemize}
        \end{itemize}
        {\scriptsize Image: An example of where the measurements can be found in the plots.}
        \end{column}

\begin{frame}{What went wrong?}{Checking the shapes of the optical elements}
    \begin{columns}
        \begin{column}{0.55\textwidth}

            \item Multiple differential equations for the shape of the RGD $\tau_D(r)$\begin{itemize}

            \end{itemize}
            \item Multiple equations for the shape of the axiparabola $s_f(r)$\begin{itemize}

                \item All are very similar shapes

        \end{itemize}

\section{Summary and Outlook}

        \item The flying focus pulse was simulated and compared in Lasy and PIConGPU

            \item So far, no test successfully showed the flying focus effect
        \end{itemize}
